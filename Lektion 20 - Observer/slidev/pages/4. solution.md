IObserver[] observers = new IObserver[]
{
    new Log(), 
    new Ui(), 
    new Mail(), 
    new Stock()
};

ISubject order = new Order();

Console.WriteLine("Register Observers");
foreach (IObserver observer in observers)
{
    order.RegisterObserver(observer);
}

order.ChangeState("With currier");
